    def forward(
    self,
    input: torch.Tensor | tuple[torch.Tensor, torch.Tensor],
    target: torch.Tensor,
) -> torch.Tensor:
    """Compute the loss of the outputs `input` w.r.t. the targets `target`.

    This is `nn.functional.gaussian_nll_loss` without its check for negative
    variances, which is data-dependent control flow that `torch.func.vmap` can't
    trace, e.g. for the per-sample gradients of the empirical Fisher. Negative
    variances are clamped at `eps` like small ones.
    """
    mean, var = as_joined_outputs(input).chunk(2, dim=-1)

    # As in torch, the clamp changes the value but passes the gradient through
    var = var + (var.clamp(min=self.eps) - var).detach()
    loss = 0.5 * (var.log() + (mean - target).square() / var)

    if self.full:
        loss = loss + log(sqrt(2 * pi))

    if self.reduction == "sum":
        return loss.sum()

    if self.reduction == "mean":
        return loss.mean()

    return loss

    Likelihood #

    Base class for likelihoods.

    A likelihood compares the model output f with the targets y. All leading dims of f but the last are batch dims, so f has shape (batch, ..., outputs).

    To add a new likelihood, subclass this, set name, and implement distribution(f), which returns the model's predictive distribution over the targets as a torch.distributions.Distribution. Its batch_shape + event_shape is the shape of the targets. All other members have defaults built on it, and you can override any of them:

    • loss_fn: the negative log-probability of the targets, summed or averaged.
    • loss_factor: \(1\), since the loss is already the negative log likelihood.
    • log_normalizer: zero.
    • loss: calls check_targets, then loss_fn.
    • check_targets: raises a ValueError unless the targets have the distribution's shape.
    • n_observations: the number of entries in the batch dims of the targets, i.e. their leading dims that match the batch dims of f.
    • inverse_link: the mean of the distribution, or the class probabilities of a Categorical.
    • sample: a draw from the distribution.
    • output_gradient: the gradient of the negative log likelihood, by autograd.
    • output_hessian: the Fisher of the distribution w.r.t. f, see below.
    • is_gaussian: False.
    • fit_likelihood and predictive_likelihood: self.

    The default output_hessian returns the Fisher

    \[ F(f) = \mathbb{E}_{y \sim p(\cdot \mid f)}\left[g_y g_y^\top\right], \qquad g_y = \nabla_f \left(-\log p(y \mid f)\right), \]

    in place of the Hessian of the negative log likelihood. The two agree for exponential families in their natural parameters, such as the Gaussian mean and the categorical logits. Elsewhere the Hessian can be indefinite, and the GGN needs the positive semidefinite \(F\). The Fisher is exact when the distribution has a finite support to enumerate, such as categorical, Bernoulli, or binomial. Otherwise it is a Monte Carlo estimate from fisher_samples draws, taken fisher_chunk_size at a time, and a UserWarning says so. If you know the Fisher in closed form, override output_hessian to make it exact and cheap. output_hessian also gets the targets y when the caller has them, for likelihoods whose curvature depends on them, e.g. through class weights or ignored positions. It is None when the caller has no targets.

    The built-in likelihoods override all of these with closed forms and the exact torch losses. Their distribution is the model's predictive, so it does not change with options that only reweight or smooth the loss.

    Two likelihoods are equal if they have the same name and the same hyperparameters(), the options that differ from their defaults. A string equals a likelihood with that name. Internally, the library dispatches on name, so a subclass of CategoricalLikelihood that keeps its name and carries options still takes the classification code paths, even though it does not equal Likelihood.CLASSIFICATION.

    Weight-space Laplace fits a custom likelihood with the diag or full structure and a GGN or EF backend, which use its output_hessian and loss_fn. The kron structure and the Hessian backends raise a ValueError for it. The GLM predictive of a likelihood that is neither Gaussian nor categorical is the Monte Carlo average of inverse_link over the GLM predictive of the outputs. FunctionalLaplace, ELLA, and VaLLA only accept the names of the regression, classification, and reward-modeling likelihoods, but a subclass that keeps one of them may override members such as output_hessian.

    def check_targets(self, f: torch.Tensor, y: torch.Tensor) -> None:
    """Raise a `ValueError` if the targets `y` do not fit the outputs `f`. By
    default, `y` must have the shape `batch_shape + event_shape` of
    `distribution(f)`.
    """
    dist = self.distribution(f)
    shape = dist.batch_shape + dist.event_shape

    if y.shape != shape:
        raise ValueError(
            f"{type(self).__name__} expects targets of shape {tuple(shape)} "
            f"for outputs of shape {tuple(f.shape)}, but the targets have shape "
            f"{tuple(y.shape)}."
        )

    def _enumerated_fisher(
    self, f: torch.Tensor, dist: torch.distributions.Distribution

    """Exact Fisher of a distribution with enumerable support.

    Parameters
    ----------
    f : torch.Tensor
        detached outputs `(batch, ..., outputs)`
    dist : torch.distributions.Distribution
        `distribution(f)`

    Returns
    -------
    F : torch.Tensor
        `(batch, ..., outputs, outputs)`
    """

    f_rep = f.expand(ys.shape[0], *f.shape).clone().requires_grad_(True)

    log_p = self.distribution(f_rep).log_prob(ys)
    (g,) = torch.autograd.grad(-log_p.sum(), f_rep)

    # independent, so the Fisher is diagonal

    return torch.einsum("s...,s...k,s...c->...kc", p_ys, g, g)

    def _monte_carlo_fisher(self, f: torch.Tensor) -> torch.Tensor:
    """Monte Carlo estimate of the Fisher from `fisher_samples` draws.

    Parameters
    ----------
    f : torch.Tensor
        detached outputs `(batch, ..., outputs)`

    Returns
    -------
    F : torch.Tensor
        `(batch, ..., outputs, outputs)`
    """
    warnings.warn(
        f"{type(self).__name__} estimates its output Hessian as a Monte Carlo "
        f"Fisher from {self.fisher_samples} samples. Override `output_hessian` "
        "with a closed form to make it exact.",
        UserWarning,
        stacklevel=3,
    )

    F = f.new_zeros(*f.shape, f.shape[-1])
    remaining = self.fisher_samples

    while remaining > 0:
        n_samples = min(self.fisher_chunk_size, remaining)
        f_rep = f.expand(n_samples, *f.shape).clone().requires_grad_(True)

        dist = self.distribution(f_rep)
        ys = dist.sample()
        (g,) = torch.autograd.grad(-dist.log_prob(ys).sum(), f_rep)

        F += torch.einsum("s...k,s...c->...kc", g, g)
        remaining -= n_samples

    GaussianLikelihood #

    GaussianLikelihood(*, sigma_noise: Tensor | None = None)

    Bases: Likelihood

    Gaussian likelihood, i.e. regression.

    With observation noise \(\sigma\), a target \(y \in \mathbb{R}^K\) given the outputs \(f\) has the likelihood

    \[ p(y \mid f) = \mathcal{N}(y \mid f, \sigma^2 I). \]

    The loss is the summed MSE, so for \(N\) observations the negative log likelihood is

    \[ -\log p(\mathcal{D} \mid f) = \frac{1}{\sigma^2} \cdot \frac{1}{2} \sum_{n=1}^N \lVert y_n - f_n \rVert^2 + N K \log\left(\sigma \sqrt{2 \pi}\right), \]

    which gives the loss factor \(\tfrac{1}{2}\) and the normalizer \(N K \log(\sigma \sqrt{2\pi})\). The targets have as many dims as the outputs, and every entry of the targets but the last dim is an observation.

    By default, the noise enters through sigma_noise of the Laplace approximation, so the other quantities assume \(\sigma = 1\): the output Hessian is \(I\), the inverse link is the identity, samples are \(y \sim \mathcal{N}(f, I)\), and the output gradient is \(f - y\). A float sigma_noise gives a normalizer with the default dtype.

    For a known noise std per output, \(\sigma \in \mathbb{R}^K\), pass a 1-D tensor as sigma_noise. The likelihood is then \(\mathcal{N}(y \mid f, \operatorname{diag}(\sigma^2))\) with the loss

    \[ \frac{1}{2} \sum_{n=1}^N \sum_{k=1}^K \frac{(y_{nk} - f_{nk})^2}{\sigma_k^2}, \]

    the output Hessian \(\operatorname{diag}(\sigma^{-2})\), and the normalizer \(N \sum_k \log(\sigma_k \sqrt{2\pi})\). The samples and the output gradient use it too. The vector is fixed when a Laplace approximation is fit, and the sigma_noise of the Laplace approximation must then stay 1. It is a hyperparameter of the likelihood, so checkpoints store it.

      known noise std per output (outputs,). All entries must be positive and finite. None leaves the noise to the Laplace approximation.

    HeteroscedasticGaussianLikelihood #

    HeteroscedasticGaussianLikelihood(
    eps: float = 1e-06, full: bool = False
)

    Bases: Likelihood

    Heteroscedastic Gaussian likelihood, i.e. regression where the network also predicts the noise, as trained with nn.GaussianNLLLoss.

    The network returns a mean \(\mu(x) \in \mathbb{R}^K\) and a variance \(\sigma^2(x)\), as the pair (mean, var) that nn.GaussianNLLLoss takes. A target then has the likelihood

    \[ p(y \mid x) = \mathcal{N}\left(y \mid \mu(x), \operatorname{diag}(\sigma^2(x))\right). \]

    The variance has the shape of the mean, or it is shared by the outputs of a datum, see join_mean_variance. The targets have the shape of the mean, and every entry of the targets but the last dim is an observation. The options eps and full are those of nn.GaussianNLLLoss: the variance is clamped at eps from below, and full adds the constant \(\tfrac{1}{2} \log 2\pi\) of each output to the loss. The loss is the summed nn.GaussianNLLLoss,

    \[ \frac{1}{2} \sum_{n=1}^N \sum_{k=1}^K \left(\log \sigma_{nk}^2 + \frac{(y_{nk} - \mu_{nk})^2}{\sigma_{nk}^2}\right), \]

    which is the negative log likelihood up to the normalizer \(N K \log \sqrt{2\pi}\), or zero with full=True. So the loss factor is \(1\).

    The Laplace approximation is over the mean only, as in Immer et al. (2023): the variance is held at its value under the MAP parameters. The GGN is then

    \[ \sum_{n=1}^N J_\mu(x_n)^\top \operatorname{diag}\left(\sigma^{-2}(x_n) \right) J_\mu(x_n), \]

    and parameters that only the variance depends on get no curvature from the data, only from the prior. The predictive la(x) returns f_mean, f_alea, f_epis: the mean, the network's variance, which is the aleatoric uncertainty, and the covariance of the mean under the posterior, which is the epistemic uncertainty. The predictive variance of the targets is f_alea plus the diagonal of f_epis.

    Inside the library, the outputs are (batch, ..., 2 * K), the mean followed by the variance, and every method of this class takes them in that form. Laplace joins the pair of the network while it runs the model. The output Hessian and the output gradient are zero for the variance, and for the mean they are \(\operatorname{diag}(\sigma^{-2})\) and \((\mu - y) / \sigma^2\). inverse_link returns the mean, and the predictive likelihood is a GaussianLikelihood over it. loss_fn also takes the pair, so marglik_training can train the network with it.

    Weight-space Laplace supports the diag, full, and kron structures with the Curvlinops backends, and diag and full with any GGN or EF backend. Last-layer Laplace and the function-space methods don't support this likelihood.

    def _enumerated_fisher(
    self, f: torch.Tensor, dist: torch.distributions.Distribution
) -> torch.Tensor:
    """Exact Fisher of a distribution with enumerable support.

    Parameters
    ----------
    f : torch.Tensor
        detached outputs `(batch, ..., outputs)`
    dist : torch.distributions.Distribution
        `distribution(f)`

    Returns
    -------
    F : torch.Tensor
        `(batch, ..., outputs, outputs)`
    """

    f_rep = f.expand(ys.shape[0], *f.shape).clone().requires_grad_(True)

    log_p = self.distribution(f_rep).log_prob(ys)

    # independent, so the Fisher is diagonal

        return torch.diag_embed((p_ys * g**2).sum(0))

    return torch.einsum("s...,s...k,s...c->...kc", p_ys, g, g)

    def _monte_carlo_fisher(self, f: torch.Tensor) -> torch.Tensor:
    """Monte Carlo estimate of the Fisher from `fisher_samples` draws.

    Parameters
    ----------
    f : torch.Tensor
        detached outputs `(batch, ..., outputs)`

    Returns
    -------
    F : torch.Tensor
        `(batch, ..., outputs, outputs)`
    """
    warnings.warn(
        f"{type(self).__name__} estimates its output Hessian as a Monte Carlo "
        f"Fisher from {self.fisher_samples} samples. Override `output_hessian` "
        "with a closed form to make it exact.",
        UserWarning,
        stacklevel=3,

        F += torch.einsum("s...k,s...c->...kc", g, g)
        remaining -= n_samples

    CategoricalLikelihood #

    CategoricalLikelihood(
    weight: Tensor | None = None,
    ignore_index: int = -100,
    label_smoothing: float = 0.0,
)

    Bases: Likelihood

    Categorical likelihood with a softmax link, i.e. classification.

    With logits \(f \in \mathbb{R}^C\) and class probabilities \(p = \operatorname{softmax}(f)\), a class \(y\) has the likelihood \(p(y \mid f) = p_y\). The loss is the cross-entropy \(-\log p_y\), or \(-\sum_c y_c \log p_c\) for class probabilities \(y\), which is already the negative log likelihood. So the loss factor is \(1\) and the normalizer is zero.

    The targets are class indices (batch, ...) or class probabilities with the shape of the logits, e.g. one-hot or soft labels. As in nn.CrossEntropyLoss, class probabilities must be floating point, and integer targets with the shape of the logits raise a ValueError: call .float() on one-hot targets. For logits with extra batch dims, e.g. (B, L, C) for sequences, loss flattens the batch dims, since CrossEntropyLoss expects the class axis at dim 1. Every position is then an observation; the class dim of probability targets does not count.

    The options weight, ignore_index, and label_smoothing are those of nn.CrossEntropyLoss, with the same defaults and meaning. Let \(q\) be the one-hot index target or the probability target, smoothed as \(q \leftarrow (1 - \varepsilon)\, q + \varepsilon / C\) with \(\varepsilon\) the label smoothing. With class weights \(w\), the loss of a position is

    \[ \ell(f, y) = -m \sum_c w_c q_c \log p_c, \]

    where \(m = [y \neq \text{ignore\_index}]\) for index targets and \(m = 1\) for probability targets, as in torch. Without weights, \(w = 1\). Only positions with \(m = 1\) count as observations. With the per-position scale \(s = w^\top q\), or \(s = 1\) without weights, the output gradient is

    \[ \nabla_f \ell = m \left(s\, p - w \odot q\right), \]

    which is \(p - y\) for one-hot \(y\) and default options, and the output Hessian is

    \[ \nabla^2_f \ell = m\, s \left(\operatorname{diag}(p) - p p^\top\right). \]

    Without targets, output_hessian returns \(\operatorname{diag}(p) - p p^\top\) without \(m\) and \(s\). The weight is cast to the dtype and device of the logits wherever it is used. Class weights and label smoothing make this a generalized likelihood, so log_marginal_likelihood is no longer a true model evidence.

    distribution, the inverse link (the softmax), and sample (one-hot draws \(y \sim \operatorname{Cat}(p)\)) ignore the options, since they describe the model's predictive, which the options leave unchanged.

    Parameters
    ----------
    f : torch.Tensor
        detached outputs `(batch, ..., outputs)`
    dist : torch.distributions.Distribution
        `distribution(f)`

    Returns
    -------
    F : torch.Tensor
        `(batch, ..., outputs, outputs)`

    f_rep = f.expand(ys.shape[0], *f.shape).clone().requires_grad_(True)

    def _monte_carlo_fisher(self, f: torch.Tensor) -> torch.Tensor:
    """Monte Carlo estimate of the Fisher from `fisher_samples` draws.

    Parameters
    ----------
    f : torch.Tensor
        detached outputs `(batch, ..., outputs)`

    Returns
    -------
    F : torch.Tensor
        `(batch, ..., outputs, outputs)`
    """
    warnings.warn(
        f"{type(self).__name__} estimates its output Hessian as a Monte Carlo "
        f"Fisher from {self.fisher_samples} samples. Override `output_hessian` "
        "with a closed form to make it exact.",
        UserWarning,
        stacklevel=3,
    )

    F = f.new_zeros(*f.shape, f.shape[-1])
    remaining = self.fisher_samples

    while remaining > 0:
        n_samples = min(self.fisher_chunk_size, remaining)
        f_rep = f.expand(n_samples, *f.shape).clone().requires_grad_(True)

        dist = self.distribution(f_rep)
        ys = dist.sample()
        (g,) = torch.autograd.grad(-dist.log_prob(ys).sum(), f_rep)

        F += torch.einsum("s...k,s...c->...kc", g, g)
        remaining -= n_samples

    return F / self.fisher_samples

    BernoulliLikelihood #

    BernoulliLikelihood(
    weight: Tensor | None = None,
    pos_weight: Tensor | None = None,
)

    Bases: Likelihood

    Bernoulli likelihood with a sigmoid link, e.g. for binary or multi-label classification and per-pixel masks.

    With logits \(f \in \mathbb{R}^K\) and probabilities \(p = \sigma(f)\), every output is an independent binary target \(y_k \in \{0, 1\}\) with \(p(y_k \mid f_k) = p_k^{y_k} (1 - p_k)^{1 - y_k}\). The loss is the binary cross-entropy of nn.BCEWithLogitsLoss, which is already the negative log likelihood. So the loss factor is \(1\) and the normalizer is zero.

    The targets are floats with the shape of the logits, (batch, ..., outputs). As in torch, they may be soft labels in \([0, 1]\). The outputs are on the last dim, so an observation is one row of outputs. For logits (B, L, K), there are \(B L\) observations.

    The options weight and pos_weight are those of nn.BCEWithLogitsLoss, with the same defaults and meaning, except that they are scalars or 1-D with one entry per output. With \(w\) the weight, \(p_w\) the positive weight, both \(1\) by default, and \(a = p_w\, y + 1 - y\), the loss of an output is

    \[ \ell(f, y) = -w \left(p_w\, y \log p + (1 - y) \log (1 - p)\right). \]

    Its gradient is

    \[ \nabla_f \ell = w \left(a\, p - p_w\, y\right), \]

    which is \(p - y\) with the default options, and the output Hessian is the diagonal matrix

    \[ \nabla^2_f \ell = \operatorname{diag}\left(w\, a\, p (1 - p)\right). \]

    Without targets, output_hessian returns \(\operatorname{diag}(p (1 - p))\) without \(w\) and \(a\). The options are cast to the dtype and device of the logits wherever they are used. They make this a generalized likelihood, so log_marginal_likelihood is no longer a true model evidence.

    distribution, the inverse link (the sigmoid), and sample (draws \(y \sim \operatorname{Bernoulli}(p)\)) ignore the options, since they describe the model's predictive, which the options leave unchanged. The log_prob of the distribution rejects soft labels, so the loss and the output gradient don't use it.

    BradleyTerryLikelihood()

    Bases: Likelihood

    Bradley-Terry likelihood for reward modeling.

    The model outputs a reward \(r(x)\) per item. For a pair of items \(x_1\) and \(x_2\), the probability that \(x_1\) is preferred is

    \[ p(x_1 \succ x_2) = \frac{\exp r(x_1)}{\exp r(x_1) + \exp r(x_2)}, \]

The curvature is therefore fitted with a CategoricalLikelihood over the compared items, so the loss, the target check, the output Hessian, the inverse link, samples, and the output gradient are all categorical, and so is the count of observations: one per probability target or per index target that isn't ignored. The predictive is a GaussianLikelihood over the reward \(r(x)\).

    def get_likelihood(likelihood: Likelihood | str | nn.Module) -> Likelihood:
    """Turn a likelihood name or a torch criterion into a likelihood object.

    Parameters
    ----------
    likelihood : Likelihood or str or nn.Module
        an instance, which is returned as is, one of `'regression'`,
        `'classification'`, `'reward_modeling'`, `'bernoulli'`, and
        `'heteroscedastic_gaussian'`, or a torch criterion. `nn.MSELoss` becomes a
        `GaussianLikelihood`, `nn.CrossEntropyLoss` a `CategoricalLikelihood` with the
        criterion's `weight`, `ignore_index`, and `label_smoothing`,
        `nn.BCEWithLogitsLoss` a `BernoulliLikelihood` with its `weight` and
        `pos_weight`, and `nn.GaussianNLLLoss` a `HeteroscedasticGaussianLikelihood`
        with its `eps` and `full`. The criterion's `reduction` is ignored, since
        Laplace always sums the loss.

    Returns
    -------
    likelihood : Likelihood

    Raises
    ------
    ValueError
        if `likelihood` is neither a likelihood instance, a known name, nor one of
        the four criteria above. Subclasses of the criteria are not accepted. Also
        if the `weight` or `pos_weight` of a `nn.BCEWithLogitsLoss` has more than one
        dim.
    """
    if isinstance(likelihood, Likelihood):
        return likelihood

    if isinstance(likelihood, str):
        for cls in (
            GaussianLikelihood,
            CategoricalLikelihood,
            BradleyTerryLikelihood,
            BernoulliLikelihood,
            HeteroscedasticGaussianLikelihood,
        ):
            if cls.name == _name_of(likelihood):
                return cls()

    if type(likelihood) is nn.MSELoss:
        return GaussianLikelihood()

    if type(likelihood) is nn.GaussianNLLLoss:
        return HeteroscedasticGaussianLikelihood(
            eps=likelihood.eps, full=likelihood.full
        )

    if type(likelihood) is nn.CrossEntropyLoss:
        return CategoricalLikelihood(
            weight=likelihood.weight,
            ignore_index=likelihood.ignore_index,
            label_smoothing=likelihood.label_smoothing,
        )

    if type(likelihood) is nn.BCEWithLogitsLoss:
        return BernoulliLikelihood(
            weight=likelihood.weight, pos_weight=likelihood.pos_weight
        )

    raise ValueError(f"Invalid likelihood type {likelihood}")
